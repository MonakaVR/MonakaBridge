cmake_minimum_required(VERSION 3.20)
project(MonakaBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 add_compile_options(/utf-8)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/import_upstream.py" RESULT_VARIABLE upstream_result)
if(NOT upstream_result EQUAL 0)
 message(FATAL_ERROR "Actual upstream artifact verification failed")
endif()
set(MONAKA_BUILD_TESTS ON CACHE BOOL "Verify fixed codec" FORCE)
add_subdirectory(third_party/monaka-protocol/cpp)
add_library(monaka_bridge_core STATIC src/observation/registry.cpp src/calibration/calibration.cpp src/mapping/config.cpp src/routing/bridge.cpp src/routing/direct.cpp)
target_include_directories(monaka_bridge_core PUBLIC include PRIVATE third_party/monaka-protocol/cpp/third_party)
target_link_libraries(monaka_bridge_core PUBLIC MonakaProtocol::Codec)
add_library(monaka_bridge_transport STATIC src/routing/network.cpp)
target_include_directories(monaka_bridge_transport PUBLIC include)
if(WIN32)
 target_compile_definitions(monaka_bridge_transport PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(monaka_bridge_transport PRIVATE ws2_32 ole32)
endif()
add_executable(monaka_bridge_service tools/bridge_service.cpp)
target_link_libraries(monaka_bridge_service PRIVATE monaka_bridge_core monaka_bridge_transport)
target_include_directories(monaka_bridge_service PRIVATE third_party/monaka-protocol/cpp/third_party)
if(WIN32)
 target_compile_definitions(monaka_bridge_service PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
add_executable(monaka_bridge_config tools/bridge_config.cpp)
target_link_libraries(monaka_bridge_config PRIVATE monaka_bridge_core)
option(MB_BUILD_STEAMVR "Build the common thin driver and migrated calibrator" ${WIN32})
if(MB_BUILD_STEAMVR)
 set(OPENVR_SDK "${CMAKE_CURRENT_SOURCE_DIR}/build/openvr-sdk" CACHE PATH "Verified Valve OpenVR v2.15.6 checkout")
 execute_process(COMMAND git -C "${OPENVR_SDK}" rev-parse HEAD OUTPUT_VARIABLE openvr_revision OUTPUT_STRIP_TRAILING_WHITESPACE)
 if(NOT openvr_revision STREQUAL "0924064316de3effbcd1acf1e309182a2deb1c05")
  message(FATAL_ERROR "Supply OpenVR v2.15.6 at exact verified revision 0924064316de3effbcd1acf1e309182a2deb1c05")
 endif()
 add_library(openvr_api STATIC IMPORTED GLOBAL)
 set_target_properties(openvr_api PROPERTIES IMPORTED_LOCATION "${OPENVR_SDK}/lib/win64/openvr_api.lib" INTERFACE_INCLUDE_DIRECTORIES "${OPENVR_SDK}/headers")
 add_library(monaka_bridge_driver SHARED outputs/steamvr/driver/hmd_driver_factory.cpp outputs/steamvr/driver/device_provider.cpp outputs/steamvr/driver/tracker_device.cpp)
 target_link_libraries(monaka_bridge_driver PRIVATE monaka_bridge_core monaka_bridge_transport openvr_api)
 target_compile_definitions(monaka_bridge_driver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 set(package "${CMAKE_BINARY_DIR}/steamvr/monaka_bridge")
 set_target_properties(monaka_bridge_driver PROPERTIES OUTPUT_NAME driver_monaka_bridge PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${package}/bin/win64")
 foreach(configuration RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
  set_target_properties(monaka_bridge_driver PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${package}/bin/win64")
 endforeach()
 add_custom_command(TARGET monaka_bridge_driver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/outputs/steamvr/package/driver.vrdrivermanifest" "${package}/driver.vrdrivermanifest" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OPENVR_SDK}/LICENSE" "${package}/OpenVR-LICENSE")
 add_executable(monaka_bridge_calibrator tools/steamvr_calibrator.cpp)
 target_link_libraries(monaka_bridge_calibrator PRIVATE monaka_bridge_core openvr_api)
 add_custom_command(TARGET monaka_bridge_calibrator POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OPENVR_SDK}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:monaka_bridge_calibrator>/openvr_api.dll")
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(bridge_tests tests/bridge_tests.cpp)
 target_link_libraries(bridge_tests PRIVATE monaka_bridge_core monaka_bridge_transport)
 target_compile_definitions(bridge_tests PRIVATE ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME bridge_core COMMAND bridge_tests)
 add_test(NAME upstream_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/import_upstream.py")
 add_executable(edge_tests tests/edge_tests.cpp)
 target_link_libraries(edge_tests PRIVATE monaka_bridge_core)
 target_compile_definitions(edge_tests PRIVATE ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME bridge_edges COMMAND edge_tests)
 add_executable(bridge_performance tests/performance.cpp)
 target_link_libraries(bridge_performance PRIVATE monaka_bridge_core)
 if(MB_BUILD_STEAMVR)
  add_executable(direct_regression tests/direct_regression.cpp build/upstream/task2/outputs/steamvr/driver/tracker_device.cpp)
  target_include_directories(direct_regression PRIVATE "${OPENVR_SDK}/headers" outputs/steamvr/driver build/upstream/task2/include build/upstream/task2/outputs/steamvr/driver)
  target_link_libraries(direct_regression PRIVATE monaka_bridge_core)
  add_executable(direct_udp_consumer tests/direct_udp_consumer.cpp)
  target_include_directories(direct_udp_consumer PRIVATE "${OPENVR_SDK}/headers" outputs/steamvr/driver third_party/monaka-protocol/cpp/third_party)
  target_link_libraries(direct_udp_consumer PRIVATE monaka_bridge_core monaka_bridge_transport)
  add_test(NAME separate_process_udp COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_regression.py" --bin "$<TARGET_FILE_DIR:monaka_bridge_service>")
  set_tests_properties(separate_process_udp PROPERTIES TIMEOUT 40)
  add_test(NAME direct_regression COMMAND direct_regression)
  set_tests_properties(direct_regression PROPERTIES TIMEOUT 20)
  if(MSVC)
   add_executable(old_context_repro tests/old_context_repro.cpp build/upstream/task2/outputs/steamvr/driver/tracker_device.cpp)
   target_include_directories(old_context_repro PRIVATE "${OPENVR_SDK}/headers" build/upstream/task2/include build/upstream/task2/outputs/steamvr/driver)
   target_compile_options(old_context_repro PRIVATE /Od /Zi)
   target_link_options(old_context_repro PRIVATE /DEBUG)
   target_link_libraries(old_context_repro PRIVATE dbghelp)
   add_test(NAME old_context_null_read_reproduced COMMAND old_context_repro)
   set_tests_properties(old_context_null_read_reproduced PROPERTIES TIMEOUT 20)
  endif()
 endif()
endif()
